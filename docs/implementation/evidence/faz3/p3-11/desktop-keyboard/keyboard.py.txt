import sys, json
from pathlib import Path
repo = Path(sys.argv[1]).resolve(); out = Path(sys.argv[2]); out.mkdir(parents=True, exist_ok=True)
sys.path.insert(0, str(repo / 'tools/ui-capture')); sys.dont_write_bytecode = True
import desktop
from playwright.sync_api import sync_playwright
DESC = """() => { const e = document.activeElement; if (!e) return null;
  const cs = getComputedStyle(e); const lab = e.closest('label'); const lcs = lab ? getComputedStyle(lab) : null;
  const grp = e.closest('[role=radiogroup]');
  return { tag: e.tagName, type: e.type || null, name: e.name || null, value: e.value || null, checked: e.checked ?? null,
    text: (lab ? lab.innerText : e.innerText || '').trim().slice(0, 40), group: grp ? grp.getAttribute('aria-label') : null,
    labelOutlineStyle: lcs ? lcs.outlineStyle : null, labelOutlineWidth: lcs ? lcs.outlineWidth : null,
    ownOutlineStyle: cs.outlineStyle, ownOutlineWidth: cs.outlineWidth }; }"""
STUB2 = """
(() => {
  const status = { receiver: { available: true, activeTransfers: 0 }, backup: { active: false, failed: false },
    autostart: { enabled: false, supported: true }, autostartChoice: false, canHide: true,
    environmentLimited: false, shutdownError: false, appVersion: '0.0.0' };
  const calls = [];
  Object.defineProperty(window, '__captureCalls', { value: calls });
  const blocked = (name) => async () => { calls.push(name); throw new Error('capture: bridge call outside read-only allowlist'); };
  window.kloda = {
    background: { status: async () => structuredClone(status), setAutostart: blocked('background.setAutostart'), quit: blocked('background.quit') },
    getDraft: async () => ({ intervalHours: 12, localTime: '12:00', sources: [], storage: null, storageTested: false }),
    getProgress: async () => null,
  };
})();
"""
log, checks = [], []
def ok(name, cond, detail=None):
    checks.append({'check': name, 'pass': bool(cond), 'detail': detail}); print(('PASS ' if cond else 'FAIL ') + name, detail if not cond else '')
srv = desktop._serve(repo / desktop.RENDERER)
errors = []
with sync_playwright() as p:
    b = p.chromium.launch()
    ctx = b.new_context(viewport={'width': 1440, 'height': 1000}, color_scheme='dark', reduced_motion='reduce')
    ctx.add_init_script(STUB2)
    ctx.add_init_script("localStorage.setItem('kloda.appearance', 'dark')")
    ctx.set_default_timeout(8000)
    page = ctx.new_page()
    page.on('console', lambda m: errors.append(m.text) if m.type == 'error' else None)
    page.on('pageerror', lambda e: errors.append(str(e)))
    page.goto(f'http://127.0.0.1:{srv.server_address[1]}/index.html#/settings')
    page.wait_for_selector('.appearance-settings'); page.locator('input[name=kloda-language][value=tr]').check()
    page.wait_for_function('document.fonts.status === "loaded"')
    page.evaluate("document.activeElement && document.activeElement.blur()")
    # a11y tree of both groups (role/name/checked) before interaction
    snap = {g: page.get_by_role('radiogroup', name=g).aria_snapshot() for g in ('Görünüm', 'Dil')}
    (out / 'aria-before.txt').write_text('\n---\n'.join(f'{k}\n{v}' for k, v in snap.items()))
    ok('theme group role=radiogroup name Görünüm with 3 radios', page.get_by_role('radiogroup', name='Görünüm').get_by_role('radio').count() == 3)
    ok('language group role=radiogroup name Dil with 3 radios', page.get_by_role('radiogroup', name='Dil').get_by_role('radio').count() == 3)
    names = [page.get_by_role('radiogroup', name='Görünüm').get_by_role('radio').nth(i).evaluate("e=>e.closest('label').innerText.trim()") for i in range(3)]
    ok('theme DOM/tab order Sistem, Açık, Koyu', names == ['Sistem', 'Açık', 'Koyu'], names)
    # Tab until theme radio (real keyboard)
    seen = []
    for i in range(40):
        page.keyboard.press('Tab'); d = page.evaluate(DESC); seen.append(d)
        if d and d['name'] == 'kloda-appearance': break
    (out / 'tab-sequence-to-theme.json').write_text(json.dumps(seen, ensure_ascii=False, indent=1))
    d = seen[-1]
    ok('Tab reaches theme radio group (checked = Koyu)', d['name'] == 'kloda-appearance' and d['checked'] and d['text'] == 'Koyu', d)
    ok('theme radio focus visible (outline on option)', d['labelOutlineStyle'] not in (None, 'none') and d['labelOutlineWidth'] != '0px', d)
    page.screenshot(path=str(out / 'focus-theme.png'))
    page.keyboard.press('ArrowLeft'); d = page.evaluate(DESC)
    ok('ArrowLeft selects Açık and moves focus, checked', d['text'] == 'Açık' and d['checked'], d)
    ok('theme actually applied in this context only (html class light)', page.evaluate("document.documentElement.classList.contains('light')"))
    page.keyboard.press('ArrowLeft'); d = page.evaluate(DESC)
    ok('ArrowLeft selects Sistem', d['text'] == 'Sistem' and d['checked'], d)
    page.keyboard.press('ArrowRight'); page.keyboard.press('ArrowRight'); d = page.evaluate(DESC)
    ok('ArrowRight x2 returns to Koyu (checked)', d['text'] == 'Koyu' and d['checked'], d)
    ok('exactly one theme radio checked', page.locator('input[name=kloda-appearance]:checked').count() == 1)
    # Tab to language group
    page.keyboard.press('Tab'); d = page.evaluate(DESC)
    ok('Tab from theme group reaches language group (checked = Türkçe)', d['name'] == 'kloda-language' and d['checked'] and d['text'] == 'Türkçe', d)
    ok('language radio focus visible', d['labelOutlineStyle'] not in (None, 'none') and d['labelOutlineWidth'] != '0px', d)
    page.screenshot(path=str(out / 'focus-language.png'))
    page.keyboard.press('ArrowRight'); d = page.evaluate(DESC)
    ok('ArrowRight selects English, checked', d['text'] == 'English' and d['checked'], d)
    ok('document language changed in this context (en)', page.evaluate("document.documentElement.lang") == 'en', page.evaluate("document.documentElement.lang"))
    page.keyboard.press('ArrowLeft'); d = page.evaluate(DESC)
    ok('ArrowLeft returns to Türkçe', d['text'] == 'Türkçe' and d['checked'], d)
    ok('document language back to tr', page.evaluate("document.documentElement.lang") == 'tr')
    # Tab on to Quit, do not activate
    reach = None
    for i in range(10):
        page.keyboard.press('Tab'); e = page.evaluate(DESC)
        if e and e['tag'] == 'BUTTON' and 'çık' in e['text']: reach = e; break
    ok('Tab reaches bottom Quit button (role button, name Kloda’dan çık)', reach is not None, reach)
    if reach:
        ok('Quit button has visible focus', reach['ownOutlineStyle'] != 'none' and reach['ownOutlineWidth'] != '0px', reach)
        btn = page.get_by_role('button', name='Kloda’dan çık'); ok('exactly one Quit button by role/name', btn.count() == 1)
    page.screenshot(path=str(out / 'focus-quit.png'))
    calls = page.evaluate('[...window.__captureCalls]')
    ok('no bridge call outside read-only allowlist (Quit never activated)', calls == [], calls)
    ok('no console errors', errors == [], errors)
    ctx.close(); b.close()
srv.shutdown(); srv.server_close()
(out / 'checks.json').write_text(json.dumps(checks, ensure_ascii=False, indent=1))
fails = [c for c in checks if not c['pass']]
print(f'{len(checks)-len(fails)}/{len(checks)} checks passed'); sys.exit(1 if fails else 0)
