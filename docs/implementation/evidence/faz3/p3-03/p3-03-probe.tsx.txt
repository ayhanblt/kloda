// P3-03 single-use probe (temporary, removed after the run). Reports counts only, never values.
import { registerRootComponent } from 'expo';
import { requireOptionalNativeModule } from 'expo-modules-core';
import { Platform, Text } from 'react-native';
import { useEffect } from 'react';
import { smb } from '../src/shared/api/smb';
import { sftp } from '../src/shared/api/sftp';
import { ftps } from '../src/shared/api/ftps';

const uuid = /^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/i;
const factories = [
  { protocol: 'smb', module: 'KlodaSmb', call: () => smb.newRequest() },
  { protocol: 'sftp', module: 'KlodaSftp', call: () => sftp.newRequest() },
  { protocol: 'ftps', module: 'KlodaFtps', call: () => ftps.newRequest() },
];

const all: string[] = [];
function run() {
  return factories.map((f) => {
    const values: string[] = [];
    let error: string | null = null;
    try { for (let i = 0; i < 2; i++) values.push(f.call()); } catch (e) { error = e instanceof Error ? e.message.slice(0, 40) : 'non-error'; }
    all.push(...values);
    return { protocol: f.protocol, nativeModulePresent: requireOptionalNativeModule(f.module) != null, calls: values.length,
      valid: values.filter((v) => typeof v === 'string' && uuid.test(v)).length, unique: new Set(values).size, error };
  });
}

function Probe() {
  useEffect(() => {
    const results = run();
    void fetch(`http://127.0.0.1:${process.env.EXPO_PUBLIC_P303_PORT}/report`, {
      method: 'POST', body: JSON.stringify({ platform: Platform.OS, results, totalValues: all.length, totalUnique: new Set(all).size }),
    }).catch(() => undefined);
  }, []);
  return <Text>p3-03</Text>;
}
registerRootComponent(Probe);
